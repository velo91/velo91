### INSTALL YARA
sudo apt update
sudo apt install -y yara
yara --version

# buat rule sederhana untuk webshell
cd /tmp
nano webshell.yar
# isi:
rule PHP_WebShell_Simple {
    meta:
        description = "Webshell sederhana"
        severity = "high"
    strings:
        $cmd1 = "system("
        $cmd2 = "exec("
        $cmd3 = "shell_exec("
        $cmd4 = "passthru("
        $cmd5 = "eval("
        $cmd6 = "base64_decode("
    condition:
        any of them
}
# scan dengan yara
yara -r /tmp/webshell.yar /var/www

# mendapatkan rule malware
# beberapa sumber rule yang banyak digunakan:
    Neo23x0 Signature Base (berisi banyak rule YARA untuk malware, webshell, ransomware, dll.)
        sudo mkdir -p /opt/yara-rules
        cd /opt/yara-rules
        sudo git clone https://github.com/Neo23x0/signature-base.git
        # cara update rule:
        cd /opt/yara-rules/signature-base && git pull
    YARA-Rules Project
        sudo mkdir -p /opt/yara-rules
        cd /opt/yara-rules
        sudo git clone https://github.com/Yara-Rules/rules.git yara-rules
        # cara update rule:
        cd /opt/yara-rules/yara-rules && git pull

# atau cara update rule otomatis bisa juga dengan
cd /opt/yara-rules
git clone https://github.com/Neo23x0/signature-base.git
# kemudian scan dengan Signature Base:
yara -r /opt/yara-rules/signature-base/yara /var/www
# atau scan dengan YARA-Rules (hanya webshell):
yara -r /opt/yara-rules/yara-rules/webshells /var/www
# atau scan dengan YARA-Rules (hanya malware):
yara -r /opt/yara-rules/yara-rules/malware /var/www
