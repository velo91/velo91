cd ~/php-base
nano Dockerfile



----------------------------------------------------------
FROM php:8.5-fpm

ADD --chmod=0755 https://github.com/mlocati/docker-php-extension-installer/releases/latest/download/install-php-extensions /usr/local/bin/

RUN install-php-extensions \
    bcmath \
    bz2 \
    exif \
    gd \
    intl \
    zip \
    soap \
    xsl \
    mysqli \
    pdo_mysql \
    igbinary \
    msgpack \
    redis \
    memcached \
    imagick \
    oauth \
    uploadprogress \
    decimal

RUN apt-get update \
    && apt-get install -y --no-install-recommends \
        libheif1 \
        libheif-plugin-libde265 \
        libheif-plugin-x265 \
        libheif-plugin-aomdec \
        libheif-plugin-aomenc \
        libmagickcore-7.q16-10-extra \
    && rm -rf /var/lib/apt/lists/*
----------------------------------------------------------



# Build image dengan tag rilis
docker build -t php-base:8.5-fpm-custom .



# Verifikasi versi PHP
docker run --rm php-base:8.5-fpm-custom php -v



# Cek ekstensi terpasang
docker run --rm php-base:8.5-fpm-custom php -m



# Cek dukungan HEIC/HEIF
docker run --rm php-base:8.5-fpm-custom php -r 'echo Imagick::getVersion()["versionString"], "\n"; print_r(Imagick::queryFormats("HEI*"));'



# Pastikan sudah login akun docker hub
docker info | grep Username
# Kalau belum login, silakan login
# Kalau belum ada akun docker hub silakan daftar di https://app.docker.com/login
# Buat PAT => Personal Access Tokens (read & write) di https://app.docker.com/settings
# Lalu login pakai username akun docker hub
docker login -u <USERNAME>
# Isikan PAT hingga Login Succeeded
# Cek info login akun docker hub
docker info | grep Username



# Push ke Docker Hub agar image tidak hilang
docker tag php-base:8.5-fpm-custom docker.io/<USERNAME>/php-base:8.5-fpm-custom
docker push docker.io/<USERNAME>/php-base:8.5-fpm-custom



# Update compose di Dokploy
# Buka aplikasi → tab General/Compose → Ganti satu baris di service php:
php:
  image: docker.io/<USERNAME>/php-base:8.5-fpm-custom
# Simpan, lalu klik Deploy
# Setelah deploy berhasil, restart container nginx jika perlu



# Verifikasi setelah deploy di container produksi
docker ps --format 'table {{.Names}}\t{{.Image}}\t{{.Status}}'
docker exec NAMA_CONTAINER_PHP php -v
docker exec NAMA_CONTAINER_PHP php -m
docker exec NAMA_CONTAINER_PHP php -r 'echo Imagick::getVersion()["versionString"], "\n"; print_r(Imagick::queryFormats("HEI*"));'
docker logs --since 5m NAMA_CONTAINER_PHP



# ⭐⭐⭐ Auto rebuild mingguan via cron (patch PHP 8.5 versi terbaru otomatis)
sudo crontab -e
# Paste ini untuk setup cron untuk rebuild otomatis setiap hari Minggu jam 00:00
0 0 * * 0 cd ~/php-base && docker build -t php-base:8.5-fpm-custom . && docker tag php-base:8.5-fpm-custom docker.io/ubbg/php-base:8.5-fpm-custom && docker push docker.io/ubbg/php-base:8.5-fpm-custom
# Cek cron terpasang
crontab -l



# Opsional: Cek image yang tidak dipakai lagi
docker ps -a --filter ancestor=NAMA_CONTAINER_PHP --format "table {{.Names}}\t{{.Status}}"
# Opsional: Jika hasilnya kosong maka aman dihapus, cara hapus image:
docker rmi NAMA_CONTAINER_PHP 2>/dev/null || true
